// texture et sprite de Document.png
sfTexture* tileTexture;
sfSprite* tileSprite;
sfIntRect tileRect = { 0, 0, 32, 32 }; 
extern sfVector2f position;

// texture et sprite de cave_objects
sfTexture* caveTexture;
sfSprite* caveSprite;

float keyTimer = 0.0f; 

// view
extern sfVector2f posEditorView;
sfView* view;

int map[NB_LAYERS][MAP_HEIGHT][MAP_WIDTH];
int currentLayer = 0; // couche sur laquelle on dessine en EDITOR
int tileTypeX = 0;    // colonne de la tile sélectionnée dans son tileset
int tileTypeY = 0;    // ligne de la tile sélectionnée dans son tileset

// sélection des tileSets
int selTileset = 0;     // 0 = Document.png, 1 = cave_objects.png
int selW = 1, selH = 1; // taille de la sélection en tiles (4x4 = objet de 128 px)

int caveCols = 0, caveRows = 0;


// palette de sélection (panneau de droite en EDITOR)
sfView* paletteTile = NULL;     // caméra de la palette (jamais zoomée)
sfView* tabView = NULL;         // caméra de la barre d'onglets
sfRectangleShape* paletteBg = NULL;      // fond de la palette
sfRectangleShape* selectionShape = NULL; // contour jaune de la sélection
sfRectangleShape* tabShape = NULL;       // rectangle d'un onglet
sfFont* font = NULL;
sfText* tabText = NULL;                  // texte des onglets (NULL si la police est introuvable)
float paletteScroll = 0.0f;  // défilement vertical
float paletteScrollX = 0.0f; // défilement horizontal
int currentTab = 0;          // 0 = Doc, 1 = 32px, 2 = 64px, 3 = 128px, 4 = autres
const char* tabNames[NB_TABS] = { "Doc", "32", "64", "128", "Autres" };
bool graphicsReady = false;  // évite de tout charger deux fois


// objet de plusieurs tiles déclaré à la main : position et taille en TILES (pixels / 32)
typedef struct { int x, y, w, h; } TileObject;
TileObject caveObjects[] =
{
	{ 0, 0, 4, 4 }, // exemple : statue de 128x128 en haut à gauche
	// { x, y, largeur, hauteur },
};
#define NB_CAVE_OBJECTS (int)(sizeof(caveObjects) / sizeof(caveObjects[0]))

// objet tel qu'il est affiché dans la palette
typedef struct
{
	int x, y, w, h;  // position et taille dans cave_objects.png (en tiles)
	int tab;         // onglet auquel il appartient
	float px, py;    // position d'affichage dans la palette (en pixels)
} PaletteObject;

static PaletteObject objects[MAX_OBJECTS];
static int nbObjects = 0;
static float tabHeight[NB_TABS]; // hauteur du contenu de chaque onglet (calculée une seule fois)

// onglet d'un objet selon sa plus grande dimension (en tiles)
static int tabFromSize(int w, int h)
{
	int m = w > h ? w : h;
	if (m == 1) return 1; // 32 px
	if (m == 2) return 2; // 64 px
	if (m == 4) return 3; // 128 px
	return 4;             // autres tailles
}

// ajoute un objet à la liste
static void addObject(int x, int y, int w, int h)
{
	if (nbObjects >= MAX_OBJECTS) return;
	objects[nbObjects++] = (PaletteObject){ x, y, w, h, tabFromSize(w, h), 0.0f, 0.0f };
}

// largeur de la caméra de la palette (en pixels du tileset)
static float paletteViewWidth(void)
{
	return sfTexture_getSize(tileTexture).x;
}

// range les objets de chaque onglet en grille et mémorise la hauteur de chaque onglet
static void layoutAllTabs(void)
{
	float viewW = paletteViewWidth();
	tabHeight[0] = 0.0f; // l'onglet Doc affiche Document.png tel quel

	for (int tab = 1; tab < NB_TABS; tab++)
	{
		// taille de la plus grande case de l'onglet
		int cell = 0;
		for (int i = 0; i < nbObjects; i++)
		{
			if (objects[i].tab != tab) continue;
			int s = (objects[i].w > objects[i].h ? objects[i].w : objects[i].h) * TILE_PX;
			if (s > cell) cell = s;
		}
		if (cell == 0) { tabHeight[tab] = 0.0f; continue; }

		float step = cell + 4.0f;          // case + petit espace
		int cols = (int)(viewW / step);    // nombre de colonnes qui tiennent dans la palette
		if (cols < 1) cols = 1;

		int n = 0;
		for (int i = 0; i < nbObjects; i++)
		{
			if (objects[i].tab != tab) continue;
			objects[i].px = (n % cols) * step; // colonne dans la grille
			objects[i].py = (n / cols) * step; // ligne dans la grille
			n++;
		}
		tabHeight[tab] = ((n + cols - 1) / cols) * step; // nombre de lignes * hauteur d'une ligne
	}
}

// construit la liste : objets déclarés dans caveObjects + toutes les cases restantes en 1x1
static void buildObjects(void)
{
	static bool covered[MAX_OBJECTS]; // cases de cave_objects déjà prises par un objet
	nbObjects = 0;
	if (caveCols * caveRows > MAX_OBJECTS) return; // tileset trop grand : augmenter MAX_OBJECTS

	for (int i = 0; i < caveCols * caveRows; i++) covered[i] = false;

	for (int i = 0; i < NB_CAVE_OBJECTS; i++)
	{
		TileObject d = caveObjects[i];
		addObject(d.x, d.y, d.w, d.h);

		for (int j = d.y; j < d.y + d.h; j++)      // marque les cases occupées par l'objet
			for (int k = d.x; k < d.x + d.w; k++)
				if (j < caveRows && k < caveCols) covered[j * caveCols + k] = true;
	}

	for (int j = 0; j < caveRows; j++)             // les cases libres deviennent des tiles de 32 px
		for (int k = 0; k < caveCols; k++)
			if (!covered[j * caveCols + k]) addObject(k, j, 1, 1);

	layoutAllTabs();
}

// numéro stocké dans map pour une tile (tileset, colonne, ligne)
static int makeTileId(int tileset, int col, int row)
{
	return (tileset == 0) ? col + DOC_COLS * row
		: CAVE_OFFSET + row * caveCols + col;
}

// opération inverse : retrouve tileset, colonne et ligne à partir d'un numéro de map
static void decodeTileId(int id, int* tileset, int* col, int* row)
{
	if (id >= CAVE_OFFSET)
	{
		id -= CAVE_OFFSET;
		*tileset = 1;
		*col = id % caveCols;
		*row = id / caveCols;
	}
	else
	{
		*tileset = 0;
		*col = id % DOC_COLS;
		*row = id / DOC_COLS;
	}
}

// dessine une tile 32x32 d'un des deux tilesets (alpha 255 = opaque)
static void drawTileId(sfRenderWindow* _window, int tileset, int col, int row, sfVector2f pos, unsigned char alpha)
{
	sfSprite* s = (tileset == 0) ? tileSprite : caveSprite;
	sfSprite_setTextureRect(s, (sfIntRect) { { col* TILE_PX, row* TILE_PX }, { TILE_PX, TILE_PX } });
	sfSprite_setPosition(s, pos);

	if (alpha != 255) sfSprite_setColor(s, (sfColor) { 255, 255, 255, alpha }); // transparence seulement si besoin
	sfRenderWindow_drawSprite(_window, s, NULL);
	if (alpha != 255) sfSprite_setColor(s, sfWhite);                            // remet l'opacité normale
}

// case de la map sous la souris (tient compte du zoom et du déplacement de editorView)
static bool getMouseTile(sfRenderWindow* _window, int* tx, int* ty)
{
	sfVector2i m = sfMouse_getPositionRenderWindow(_window);
	sfVector2f p = sfRenderWindow_mapPixelToCoords(_window, m, editorView); // pixel écran -> coordonnées de la map
	if (p.x < 0 || p.y < 0) return false;
	*tx = (int)(p.x / TILE_PX);
	*ty = (int)(p.y / TILE_PX);
	return *tx < MAP_WIDTH && *ty < MAP_HEIGHT;
}

// true si la souris est dans la bande de droite (palette + onglets)
static bool mouseOnPalette(sfRenderWindow* _window)
{
	sfVector2u winSize = sfRenderWindow_getSize(_window);
	sfVector2i m = sfMouse_getPositionRenderWindow(_window);
	return m.x >= winSize.x * (1.0f - PALETTE_RATIO);
}

////                  PALETTE                    ////
/////////////////////////////////////////////////////

// règle la caméra de la palette (taille, défilement) selon l'onglet actif
static void setupPaletteView(sfRenderWindow* _window)
{
	sfVector2u winSize = sfRenderWindow_getSize(_window);
	sfVector2u docSize = sfTexture_getSize(tileTexture);

	float viewW = paletteViewWidth();
	// même proportions que la zone à l'écran (sinon l'image serait étirée), barre d'onglets exclue
	float viewH = viewW * (winSize.y * (1.0f - TAB_RATIO)) / (winSize.x * PALETTE_RATIO);

	// taille du contenu : Document.png entier, ou grille des objets de l'onglet
	float contentW = (currentTab == 0) ? (float)docSize.x : viewW;
	float contentH = (currentTab == 0) ? (float)docSize.y : tabHeight[currentTab];

	// on empêche de défiler au-delà du contenu
	float maxScrollX = contentW > viewW ? contentW - viewW : 0.0f;
	float maxScrollY = contentH > viewH ? contentH - viewH : 0.0f;
	if (paletteScrollX > maxScrollX) paletteScrollX = maxScrollX;
	if (paletteScrollX < 0.0f) paletteScrollX = 0.0f;
	if (paletteScroll > maxScrollY) paletteScroll = maxScrollY;
	if (paletteScroll < 0.0f) paletteScroll = 0.0f;

	sfView_setSize(paletteTile, (sfVector2f) { viewW, viewH });
	// décaler le centre de la caméra = faire défiler le contenu
	sfView_setCenter(paletteTile, (sfVector2f) { viewW / 2.0f + paletteScrollX, viewH / 2.0f + paletteScroll });
}

// sélectionne la tile (ou l'objet entier) sous la souris dans la palette
static void selectFromPalette(sfRenderWindow* _window)
{
	setupPaletteView(_window); // la caméra doit être à jour avant de convertir la souris
	sfVector2i m = sfMouse_getPositionRenderWindow(_window);
	sfVector2f p = sfRenderWindow_mapPixelToCoords(_window, m, paletteTile); // pixel écran -> coordonnées de la palette
	if (p.x < 0 || p.y < 0) return;

	if (currentTab == 0) // onglet Doc : Document.png affiché tel quel
	{
		sfVector2u docSize = sfTexture_getSize(tileTexture);
		int col = (int)(p.x / TILE_PX), row = (int)(p.y / TILE_PX);
		if (p.y < docSize.y && col < (int)(docSize.x / TILE_PX))
		{
			selTileset = 0;
			tileTypeX = col;
			tileTypeY = row;
			selW = selH = 1;
		}
		return;
	}

	// autres onglets : on cherche l'objet dessiné sous la souris
	for (int i = 0; i < nbObjects; i++)
	{
		PaletteObject* o = &objects[i];
		if (o->tab != currentTab) continue;
		if (p.x >= o->px && p.x < o->px + o->w * TILE_PX &&
			p.y >= o->py && p.y < o->py + o->h * TILE_PX)
		{
			selTileset = 1;
			tileTypeX = o->x;
			tileTypeY = o->y;
			selW = o->w;
			selH = o->h;
			return;
		}
	}
}

// pose la sélection sur la couche active, coin haut-gauche à la case (mx, my)
static void placeSelection(int mx, int my)
{
	for (int j = 0; j < selH; j++)
	{
		for (int i = 0; i < selW; i++)
		{
			int x = mx + i, y = my + j;
			if (x >= MAP_WIDTH || y >= MAP_HEIGHT) continue; // dépasse de la map : ignoré
			map[currentLayer][y][x] = makeTileId(selTileset, tileTypeX + i, tileTypeY + j);
		}
	}
}

// charge textures, sprites, palette et formes de l'interface (une seule fois)
static void initGraphics(void)
{
	if (graphicsReady) return;
	graphicsReady = true;

	// Document.png en premier : la palette en a besoin pour calculer sa largeur
	tileTexture = sfTexture_createFromFile(TEXTURE_PATH"/tileset/Document.png", NULL);
	tileSprite = sfSprite_create(tileTexture);

	caveTexture = sfTexture_createFromFile(TEXTURE_PATH"/tileset/cave_object.png", NULL);
	caveSprite = sfSprite_create(caveTexture);
	caveCols = sfTexture_getSize(caveTexture).x / TILE_PX;
	caveRows = sfTexture_getSize(caveTexture).y / TILE_PX;
	buildObjects(); // liste et range les objets de la palette

	// caméra de la palette : le viewport l'affiche à droite, sous la barre d'onglets
	paletteTile = sfView_create();
	sfView_setViewport(paletteTile, (sfFloatRect) { { 1.0f - PALETTE_RATIO, TAB_RATIO }, { PALETTE_RATIO, 1.0f - TAB_RATIO } });

	// caméra de la barre d'onglets : bande tout en haut de la palette
	tabView = sfView_create();
	sfView_setViewport(tabView, (sfFloatRect) { { 1.0f - PALETTE_RATIO, 0.0f }, { PALETTE_RATIO, TAB_RATIO } });

	// formes de l'interface
	paletteBg = sfRectangleShape_create();
	sfRectangleShape_setFillColor(paletteBg, (sfColor) { 30, 30, 30, 255 });

	selectionShape = sfRectangleShape_create();
	sfRectangleShape_setFillColor(selectionShape, sfTransparent);
	sfRectangleShape_setOutlineColor(selectionShape, sfYellow);
	sfRectangleShape_setOutlineThickness(selectionShape, 1.0f);

	tabShape = sfRectangleShape_create();

	// texte des onglets : si la police est introuvable, les onglets s'affichent sans texte
	font = sfFont_createFromFile(FONT_PATH);
	if (font)
	{
		tabText = sfText_create(font);
		sfText_setCharacterSize(tabText, 14);
	}
}

// couches 1 et 2 à "vide" pour que la couche 0 reste visible (la couche 0 est remplie par le chargement)
static void clearUpperLayers(void)
{
	for (int l = 1; l < NB_LAYERS; l++)
		for (int y = 0; y < MAP_HEIGHT; y++)
			for (int x = 0; x < MAP_WIDTH; x++)
				map[l][y][x] = EMPTY_TILE;
}

// initialise la map depuis le fichier .txt
void initMap()
{
	initGraphics();
	clearUpperLayers();

	FILE* f = fopen(MAP_TXT_PATH, "rt");
	if (!f) return; // pas de fichier : il sera créé à la première sauvegarde

	// même ordre que saveMap : i = ligne du fichier (< MAP_WIDTH), j = valeur dans la ligne (< MAP_HEIGHT)
	for (int l = 0; l < NB_LAYERS; l++)
	{
		for (int i = 0; i < MAP_WIDTH; i++)
		{
			for (int j = 0; j < MAP_HEIGHT; j++)
			{
				int v;
				// fscanf saute tout seul les espaces et retours à la ligne
				// fichier fini (ancienne map à une seule couche) : les autres couches restent vides
				if (fscanf(f, "%d", &v) != 1)
				{
					fclose(f);
					return;
				}
				map[l][j][i] = v;
			}
		}
	}
	fclose(f);
}

// initialise la map depuis le fichier .bin
void initMapBin()
{
	initGraphics();
	clearUpperLayers();

	FILE* f = fopen(MAP_BIN_PATH, "rb");
	if (!f) return;

	// lit les couches d'un coup ; un ancien fichier (1 couche) remplit seulement la couche 0
	fread(map, sizeof(int), NB_LAYERS * MAP_HEIGHT * MAP_WIDTH, f);
	fclose(f);
}

/////////////////////////////////////////////////////
////        MISE À JOUR (GAME ET EDITOR)         ////
/////////////////////////////////////////////////////

void updateMap(sfRenderWindow* _window)
{
	keyTimer += getDeltaTime();

	// touche E : bascule entre GAME et EDITOR (0.2 s entre deux appuis)
	if (sfKeyboard_isScancodePressed(sfScanE) && keyTimer >= 0.2f)
	{
		state = (state == GAME) ? EDITOR : GAME;
		keyTimer = 0.0f;
		return;
	}

	if (state != EDITOR) return; // tout le reste ne concerne que l'éditeur

	/////////////////////////////////////////////////////
	////        GESTION DES COMMANDES EDITOR         ////
	/////////////////////////////////////////////////////

	// F1, F2, F3... : choix de la couche active (les touches F sont consécutives)
	for (int l = 0; l < NB_LAYERS; l++)
		if (sfKeyboard_isKeyPressed(sfKeyF1 + l))
			currentLayer = l;

	// 1..8 : colonne de la tile / Ctrl + 1..5 : ligne de la tile (dans Document.png)
	bool ctrl = sfKeyboard_isKeyPressed(sfKeyLControl);
	for (int i = 0; i < DOC_COLS; i++)
	{
		if (!sfKeyboard_isKeyPressed(sfKeyNum1 + i)) continue;

		if (!ctrl)
		{
			tileTypeX = i;
			selTileset = 0; // le clavier ramène sur Document.png en 1x1
			selW = selH = 1;
		}
		else if (i < DOC_ROWS)
		{
			tileTypeY = i;
			selTileset = 0;
			selW = selH = 1;
		}
	}

	// défilement de la palette : PageUp/PageDown (vertical), Home/End (horizontal)
	// getDeltaTime() rend la vitesse indépendante du framerate
	if (sfKeyboard_isKeyPressed(sfKeyPageUp))   paletteScroll -= 300.0f * getDeltaTime();
	if (sfKeyboard_isKeyPressed(sfKeyPageDown)) paletteScroll += 300.0f * getDeltaTime();
	if (sfKeyboard_isKeyPressed(sfKeyHome))     paletteScrollX -= 300.0f * getDeltaTime();
	if (sfKeyboard_isKeyPressed(sfKeyEnd))      paletteScrollX += 300.0f * getDeltaTime();

	// zoom de la map avec + et - du pavé numérique
	if (sfKeyboard_isScancodePressed(sfScanNumpadPlus))
		sfView_zoom(editorView, 0.999f);
	if (sfKeyboard_isScancodePressed(sfScanNumpadMinus))
		sfView_zoom(editorView, 1.001f);

	// déplacement de la view de l'éditeur avec les flèches
	if (sfKeyboard_isScancodePressed(sfScanUp) && posEditorView.y > ((HEIGHT_WINDOW_GAME / 2)))
		posEditorView.y -= 0.4f;
	if (sfKeyboard_isScancodePressed(sfScanDown) && posEditorView.y < ((HEIGHT_WINDOW_GAME / 2)))
		posEditorView.y += 0.4f;
	if (sfKeyboard_isScancodePressed(sfScanRight) && posEditorView.x < ((WIDTH_WINDOW_GAME / 2) + WIDTH_WINDOW_GAME * PALETTE_RATIO))
		posEditorView.x += 0.4f;
	if (sfKeyboard_isScancodePressed(sfScanLeft) && posEditorView.x > ((WIDTH_WINDOW_GAME / 2)))
		posEditorView.x -= 0.4f;

	/////////////////////////////////////////////////////
	////          SOURIS : PALETTE ET MAP            ////
	/////////////////////////////////////////////////////

	bool onPalette = mouseOnPalette(_window);
	sfVector2u winSize = sfRenderWindow_getSize(_window);
	sfVector2i mouseNow = sfMouse_getPositionRenderWindow(_window);
	bool onTabBar = onPalette && mouseNow.y < winSize.y * TAB_RATIO; // souris sur la barre d'onglets

	// "static" : ces variables gardent leur valeur d'une frame à l'autre
	static bool leftWasDown = false;           // clic gauche enfoncé à la frame précédente ?
	static bool pressStartedOnPalette = false; // le clic en cours a-t-il commencé sur la palette ?
	bool leftDown = sfMouse_isButtonPressed(sfMouseLeft);

	if (leftDown && !leftWasDown) // le clic vient juste de commencer
	{
		pressStartedOnPalette = onPalette; // un clic commencé sur la palette ne posera rien sur la map

		if (onTabBar) // clic sur un onglet
		{
			float tabBarW = winSize.x * PALETTE_RATIO;
			int t = (int)((mouseNow.x - (winSize.x - tabBarW)) / (tabBarW / NB_TABS)); // numéro de l'onglet cliqué
			if (t >= 0 && t < NB_TABS)
			{
				currentTab = t;
				paletteScroll = 0.0f; // on repart en haut du nouvel onglet
				paletteScrollX = 0.0f;
			}
		}
	}
	leftWasDown = leftDown;

	// clic gauche maintenu sur le contenu de la palette : sélection
	if (leftDown && pressStartedOnPalette && onPalette && !onTabBar)
		selectFromPalette(_window);

	// souris sur la map
	int tx, ty;
	if (!onPalette && getMouseTile(_window, &tx, &ty))
	{
		// clic gauche (commencé hors palette) : pose la sélection
		if (leftDown && !pressStartedOnPalette)
			placeSelection(tx, ty);

		// clic droit : efface (tile noire sur la couche 0, case vide sur les autres)
		if (sfMouse_isButtonPressed(sfMouseRight))
			map[currentLayer][ty][tx] = (currentLayer == 0) ? BLACK_TILE : EMPTY_TILE;

		// clic milieu : copie la tile sous la souris
		if (sfMouse_isButtonPressed(sfMouseMiddle) && map[currentLayer][ty][tx] >= 0)
		{
			decodeTileId(map[currentLayer][ty][tx], &selTileset, &tileTypeX, &tileTypeY);
			selW = selH = 1;
		}
	}
}

// affiche la barre d'onglets et la palette à droite de l'écran (mode EDITOR)
void displayTilePalette(sfRenderWindow* _window)
{
	if (state != EDITOR) return;

	// on garde la caméra actuelle pour la remettre à la fin
	sfView* oldView = sfView_copy(sfRenderWindow_getView(_window));

	/////// barre d'onglets (coordonnées en pixels écran) ///////
	sfVector2u winSize = sfRenderWindow_getSize(_window);
	float tabBarW = winSize.x * PALETTE_RATIO;
	float tabBarH = winSize.y * TAB_RATIO;
	float oneTabW = tabBarW / NB_TABS;
	sfView_setSize(tabView, (sfVector2f) { tabBarW, tabBarH });
	sfView_setCenter(tabView, (sfVector2f) { tabBarW / 2.0f, tabBarH / 2.0f });
	sfRenderWindow_setView(_window, tabView);

	for (int t = 0; t < NB_TABS; t++)
	{
		sfRectangleShape_setPosition(tabShape, (sfVector2f) { t* oneTabW, 0.0f });
		sfRectangleShape_setSize(tabShape, (sfVector2f) { oneTabW - 2.0f, tabBarH });
		// onglet actif plus clair
		sfRectangleShape_setFillColor(tabShape, t == currentTab ? (sfColor) { 110, 110, 110, 255 } : (sfColor) { 50, 50, 50, 255 });
		sfRenderWindow_drawRectangleShape(_window, tabShape, NULL);

		if (tabText)
		{
			sfText_setString(tabText, tabNames[t]);
			sfText_setPosition(tabText, (sfVector2f) { t* oneTabW + 4.0f, tabBarH / 2.0f - 9.0f });
			sfRenderWindow_drawText(_window, tabText, NULL);
		}
	}

	/////// contenu de l'onglet ///////
	setupPaletteView(_window);
	sfVector2f viewSize = sfView_getSize(paletteTile);
	sfVector2f viewCenter = sfView_getCenter(paletteTile);
	sfRenderWindow_setView(_window, paletteTile);

	// fond : couvre toute la zone visible, même après défilement
	sfRectangleShape_setPosition(paletteBg, (sfVector2f) { viewCenter.x - viewSize.x / 2.0f, viewCenter.y - viewSize.y / 2.0f });
	sfRectangleShape_setSize(paletteBg, viewSize);
	sfRenderWindow_drawRectangleShape(_window, paletteBg, NULL);

	if (currentTab == 0) // onglet Doc : Document.png entier
	{
		sfVector2u docSize = sfTexture_getSize(tileTexture);
		// on remet le rectangle sur toute la texture (displayMap le réduit à 32x32)
		sfSprite_setTextureRect(tileSprite, (sfIntRect) { { 0, 0 }, { (int)docSize.x, (int)docSize.y } });
		sfSprite_setPosition(tileSprite, (sfVector2f) { 0.0f, 0.0f });
		sfRenderWindow_drawSprite(_window, tileSprite, NULL);

		if (selTileset == 0) // contour jaune sur la tile sélectionnée
		{
			sfRectangleShape_setPosition(selectionShape, (sfVector2f) { tileTypeX* (float)TILE_PX, tileTypeY* (float)TILE_PX });
			sfRectangleShape_setSize(selectionShape, (sfVector2f) { (float)TILE_PX, (float)TILE_PX });
			sfRenderWindow_drawRectangleShape(_window, selectionShape, NULL);
		}
	}
	else // autres onglets : objets de cave_objects.png rangés en grille
	{
		for (int i = 0; i < nbObjects; i++)
		{
			PaletteObject* o = &objects[i];
			if (o->tab != currentTab) continue;

			// découpe l'objet entier (w x h tiles) dans la texture et le dessine à sa place dans la grille
			sfSprite_setTextureRect(caveSprite, (sfIntRect) { { o->x* TILE_PX, o->y* TILE_PX }, { o->w * TILE_PX, o->h * TILE_PX } });
			sfSprite_setPosition(caveSprite, (sfVector2f) { o->px, o->py });
			sfRenderWindow_drawSprite(_window, caveSprite, NULL);

			// contour jaune si c'est l'objet sélectionné
			if (selTileset == 1 && o->x == tileTypeX && o->y == tileTypeY && o->w == selW && o->h == selH)
			{
				sfRectangleShape_setPosition(selectionShape, (sfVector2f) { o->px, o->py });
				sfRectangleShape_setSize(selectionShape, (sfVector2f) { o->w* (float)TILE_PX, o->h* (float)TILE_PX });
				sfRenderWindow_drawRectangleShape(_window, selectionShape, NULL);
			}
		}
	}

	// on rend la caméra d'origine
	sfRenderWindow_setView(_window, oldView);
	sfView_destroy(oldView);
}

// affiche les tiles visibles de toutes les couches, puis l'aperçu de la sélection en EDITOR
void displayMap(sfRenderWindow* _window)
{
	// zone visible de la view actuelle, convertie en numéros de cases
	// (on ne dessine que ces cases au lieu de toute la map)
	const sfView* v = sfRenderWindow_getView(_window);
	sfVector2f c = sfView_getCenter(v);
	sfVector2f s = sfView_getSize(v);
	int x0 = (int)floorf((c.x - s.x / 2.0f) / TILE_PX);
	int x1 = (int)floorf((c.x + s.x / 2.0f) / TILE_PX);
	int y0 = (int)floorf((c.y - s.y / 2.0f) / TILE_PX);
	int y1 = (int)floorf((c.y + s.y / 2.0f) / TILE_PX);

	// on reste dans les limites du tableau
	if (x0 < 0) x0 = 0;
	if (y0 < 0) y0 = 0;
	if (x1 > MAP_WIDTH - 1) x1 = MAP_WIDTH - 1;
	if (y1 > MAP_HEIGHT - 1) y1 = MAP_HEIGHT - 1;

	for (int l = 0; l < NB_LAYERS; l++) // du fond vers le dessus
	{
		for (int y = y0; y <= y1; y++)
		{
			for (int x = x0; x <= x1; x++)
			{
				int id = map[l][y][x];
				if (id < 0) continue; // case vide : on ne dessine rien, on passe à la suivante

				int tileset, col, row;
				decodeTileId(id, &tileset, &col, &row); // quel tileset, quelle colonne, quelle ligne
				drawTileId(_window, tileset, col, row, (sfVector2f) { x* (float)TILE_PX, y* (float)TILE_PX }, 255);
			}
		}
	}

	/////////////////////////////////////////////////////
	////   APERÇU DE LA SÉLECTION SOUS LE CURSEUR    ////
	/////////////////////////////////////////////////////

	if (state == EDITOR)
	{
		int tx, ty;
		// true si la souris est sur la map (pas sur la palette) et sur une case valide
		bool hovering = !mouseOnPalette(_window) && getMouseTile(_window, &tx, &ty);

		if (hovering) // aperçu semi-transparent de toute la sélection, aligné sur la grille
			for (int j = 0; j < selH; j++)
				for (int i = 0; i < selW; i++)
					drawTileId(_window, selTileset, tileTypeX + i, tileTypeY + j,
						(sfVector2f) {
					(tx + i)* (float)TILE_PX, (ty + j)* (float)TILE_PX
				}, 180);

		// cache le curseur sur la map, le montre sur la palette (modifié seulement quand ça change)
		static int lastCursor = -1;
		if (lastCursor != !hovering)
		{
			sfRenderWindow_setMouseCursorVisible(_window, !hovering);
			lastCursor = !hovering;
		}
	}
}

////      GESTION DES COLLISIONS DE LA MAP       ////
/////////////////////////////////////////////////////

// true si le point (px, py) en pixels est sur une case bloquante, ou hors de la map
static bool isSolidAt(float px, float py)
{
	int x = (int)floorf(px / TILE_PX); // floorf : un point à -0.5 donne bien -1, pas 0
	int y = (int)floorf(py / TILE_PX);

	if (x < 0 || x >= MAP_WIDTH || y < 0 || y >= MAP_HEIGHT)
		return true; // en dehors de la map = bloqué

	// une seule couche bloquante suffit (EMPTY_TILE = -1 et les tiles de cave ne bloquent pas)
	for (int l = 0; l < NB_LAYERS; l++)
		if (map[l][y][x] >= 3 && map[l][y][x] <= 39)
			return true;

	return false;
}

// true si le sprite, une fois avancé dans _direction, touche une case bloquante
bool collisionMapPlayer(sfFloatRect _sprite, Direction _direction, sfVector2f _vitesse)
{
	float dt = getDeltaTime();
	float left = _sprite.position.x, right = _sprite.position.x + _sprite.size.x;
	float top = _sprite.position.y, bottom = _sprite.position.y + _sprite.size.y;

	// deux points testés sur le côté du sprite qui avance (deux cases possibles si le sprite est entre 2 cases)
	sfVector2f a = { 0.0f, 0.0f }, b = { 0.0f, 0.0f };

	switch (_direction)
	{
	case HAUT:
		a = (sfVector2f){ left,  top - _vitesse.y * dt };
		b = (sfVector2f){ right, top - _vitesse.y * dt };
		break;
	case BAS:
		a = (sfVector2f){ left,  bottom + _vitesse.y * dt };
		b = (sfVector2f){ right, bottom + _vitesse.y * dt };
		break;
	case GAUCHE:
		a = (sfVector2f){ left - _vitesse.x * dt, top };
		b = (sfVector2f){ left - _vitesse.x * dt, bottom };
		break;
	case DROITE:
		a = (sfVector2f){ right + _vitesse.x * dt, top };
		b = (sfVector2f){ right + _vitesse.x * dt, bottom };
		break;
	default:
		return false;
	}

	return isSolidAt(a.x, a.y) || isSolidAt(b.x, b.y);
}

/////////////////////////////////////////////////////

void saveMap()
{
	FILE* f = fopen(MAP_TXT_PATH, "wt");
	if (!f) return; // dossier introuvable : on ne plante pas

	// même ordre que initMap : i = ligne du fichier (< MAP_WIDTH), j = valeur dans la ligne (< MAP_HEIGHT)
	for (int l = 0; l < NB_LAYERS; l++)
	{
		for (int i = 0; i < MAP_WIDTH; i++)
		{
			for (int j = 0; j < MAP_HEIGHT; j++)
				fprintf(f, "%d ", map[l][j][i]);
			fprintf(f, "\n");
		}
		fprintf(f, "\n"); // ligne vide entre deux couches
	}
	fclose(f);
}

void saveMapBin()
{
	FILE* f = fopen(MAP_BIN_PATH, "wb");
	if (!f) return;

	// écrit d'un coup tout le tableau (toutes les couches)
	fwrite(map, sizeof(int), NB_LAYERS * MAP_HEIGHT * MAP_WIDTH, f);
	fclose(f);
}

void destroyMap()
{
	if (tabText) { sfText_destroy(tabText);                 tabText = NULL; }
	if (font) { sfFont_destroy(font);                    font = NULL; }
	if (tabShape) { sfRectangleShape_destroy(tabShape);      tabShape = NULL; }
	if (selectionShape) { sfRectangleShape_destroy(selectionShape); selectionShape = NULL; }
	if (paletteBg) { sfRectangleShape_destroy(paletteBg);     paletteBg = NULL; }
	if (tabView) { sfView_destroy(tabView);                 tabView = NULL; }
	if (paletteTile) { sfView_destroy(paletteTile);             paletteTile = NULL; }
	if (caveSprite) { sfSprite_destroy(caveSprite);            caveSprite = NULL; }
	if (caveTexture) { sfTexture_destroy(caveTexture);          caveTexture = NULL; }
	if (tileSprite) { sfSprite_destroy(tileSprite);            tileSprite = NULL; }
	if (tileTexture) { sfTexture_destroy(tileTexture);          tileTexture = NULL; }
	graphicsReady = false;
}
